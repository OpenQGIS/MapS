5. 极简集成与直接调用 (Integration Guide)

代码零外部依赖，直接引入本地 mraph.js 即可为任意按钮赋予此动效。

Developer API 
// 1. 在现有页面中引入本地脚本
<script src="./demo/mraph.js"></script>

// 2. 为现有任意按钮绑定点击变形动效
const btnTheme = Mraph.bindButton('#theme-toggle-btn', {
  from: 'sun',
  to: 'moon',
  preset: 'snappy',   // 弹簧预设：snappy | smooth | bouncy
  ripple: true,       // 是否开启水波纹触控
  onToggle: (activeState) => {
    // 业务联动逻辑，例如切换全局浅色/深色主题
    console.log('当前图标状态:', activeState);
  }
});

// 3. 或者用于瞬时操作成功反馈 (如一键复制)
const animator = Mraph.attach(svgPathElement);
animator.morphTo('check');
setTimeout(() => animator.morphTo('copy'), 1800);
